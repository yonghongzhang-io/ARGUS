\section{Supplementary Figures and Tables}
\label{app:supp}

This appendix collects supporting material referenced from the main text:
the per-dimension risk distribution on the 27-paper real corpus
(Figure~\ref{fig:realcorpus}), and the deterministic calibration results against
the adjudicated human gold (Table~\ref{tab:calib}, Figure~\ref{fig:calib_tradeoff}),
and the agreement metrics for \sys{} versus the human gold (Table~\ref{tab:vsgold}).

\begin{figure}[t]
  \centering
  \includegraphics[width=\columnwidth]{figures/calibration_tradeoff}
  \caption{\textbf{Calibration trade-off: severity reduction versus coverage}
  against the adjudicated human gold. Demoting weak-retrieval \emph{high}s to
  \emph{medium} preserves coverage (33 answered) while cutting over-severe
  judgments from 29 to 18; abstaining gives the strongest correction
  (over-severe 6, weighted $\kappa$ $0.25$) but leaves only 13 answered cells.
  Calibration corrects severity bias; the remaining bottleneck is evidence
  grounding, not severity thresholding alone.}
  \label{fig:calib_tradeoff}
\end{figure}
